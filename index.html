/** * Backend di storage condiviso per il simulatore ERP GRS Solutions. * Sostituisce lo storage di Claude (che richiedeva un piano a pagamento) * con un Google Sheet, gratuito e raggiungibile da chiunque. * * INSTALLAZIONE (una volta sola): * 1. Vai su https://sheets.google.com e crea un foglio nuovo (vuoto va bene). * 2. Menu Estensioni → Apps Script. * 3. Cancella il contenuto di default e incolla TUTTO questo file al suo posto. * 4. Salva (icona dischetto o Ctrl+S). Dai un nome al progetto se richiesto. * 5. In alto a destra: Deploy → Nuovo deployment. * 6. Cliccando sull'icona a ingranaggio accanto a "Seleziona tipo", scegli "App web". * 7. Configura così: * - Descrizione: qualsiasi (es. "storage ERP") * - Esegui come: "Me" * - Chi ha accesso: "Chiunque" (ANCHE senza account Google — importante!) * 8. Clicca "Esegui il deployment". La prima volta Google chiederà di autorizzare * lo script: segui le richieste (potrebbe avvisare "app non verificata": è normale * per uno script personale, clicca su "Avanzate" → "Vai al progetto (non sicuro)"). * 9. Copia l'URL dell'app web che ti viene mostrato (finisce con /exec). * 10. Incolla quell'URL nel file HTML del programma, al posto di * 'INSERISCI_QUI_URL_APPS_SCRIPT' (cerca APPS_SCRIPT_URL vicino all'inizio del codice). * * Se in futuro modifichi questo script, dopo aver salvato devi fare un NUOVO deployment * (Deploy → Gestisci deployment → matita di modifica → Versione: Nuova versione → Deploy) * altrimenti le modifiche non hanno effetto sull'URL già in uso. */ const NOME_FOGLIO = 'Storage'; function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(NOME_FOGLIO); if (!sheet) { sheet = ss.insertSheet(NOME_FOGLIO); sheet.appendRow(['chiave', 'valore']); } return sheet; } // Cerca una chiave nella colonna A. Ritorna il numero di riga (1-based) o -1 se non trovata. function trovaRiga_(sheet, chiave) { const valori = sheet.getRange(1, 1, sheet.getLastRow() || 1, 1).getValues(); for (let i = 1; i < valori.length; i++) { // salta l'intestazione (riga 0) if (valori[i][0] === chiave) return i + 1; } return -1; } function doGet(e) { const azione = e.parameter.action; const chiave = e.parameter.key; let risultato; try { const sheet = getSheet_(); if (azione === 'get') { const riga = trovaRiga_(sheet, chiave); if (riga === -1) { risultato = { found: false, value: null }; } else { risultato = { found: true, value: sheet.getRange(riga, 2).getValue() }; } } else if (azione === 'getPrefix') { // Legge in UN SOLO colpo tutte le chiavi che iniziano con "prefix" — usato per // caricare tutti i dati di un corso con una sola richiesta invece di decine. const prefisso = e.parameter.prefix || ''; const dati = sheet.getDataRange().getValues(); const righeTrovate = []; for (let i = 1; i < dati.length; i++) { if (String(dati[i][0]).indexOf(prefisso) === 0) { righeTrovate.push({ key: dati[i][0], value: dati[i][1] }); } } risultato = { rows: righeTrovate }; } else { risultato = { error: 'azione GET sconosciuta: ' + azione }; } } catch (err) { risultato = { error: String(err) }; } return ContentService.createTextOutput(JSON.stringify(risultato)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const lock = LockService.getScriptLock(); let risultato; try { lock.waitLock(10000); // aspetta fino a 10s se un altro utente sta scrivendo nello stesso momento const corpo = JSON.parse(e.postData.contents); const sheet = getSheet_(); if (corpo.action === 'set') { const riga = trovaRiga_(sheet, corpo.key); if (riga === -1) { sheet.appendRow([corpo.key, corpo.value]); } else { sheet.getRange(riga, 2).setValue(corpo.value); } risultato = { ok: true }; } else if (corpo.action === 'delete') { const riga = trovaRiga_(sheet, corpo.key); if (riga !== -1) sheet.deleteRow(riga); risultato = { ok: true }; } else if (corpo.action === 'appendIfAbsent') { // Operazione ATOMICA (protetta dal lock): legge la lista, aggiunge l'elemento // SOLO se non è già presente (in base a matchField/matchValue), riscrive. // Tutto dentro lo stesso lock: elimina alla radice la perdita di dati per // scritture quasi contemporanee di più corsisti sulla stessa lista condivisa. const riga = trovaRiga_(sheet, corpo.key); let lista = []; if (riga !== -1) { try { lista = JSON.parse(sheet.getRange(riga, 2).getValue()); } catch (e2) { lista = []; } } const giaPresente = lista.some(el => el[corpo.matchField] === corpo.matchValue); if (!giaPresente) { lista.push(corpo.item); const nuovoValore = JSON.stringify(lista); if (riga === -1) sheet.appendRow([corpo.key, nuovoValore]); else sheet.getRange(riga, 2).setValue(nuovoValore); } risultato = { ok: true, value: JSON.stringify(lista) }; } else if (corpo.action === 'removeMatching') { // Operazione ATOMICA analoga, per rimuovere un elemento da una lista condivisa. const riga = trovaRiga_(sheet, corpo.key); let lista = []; if (riga !== -1) { try { lista = JSON.parse(sheet.getRange(riga, 2).getValue()); } catch (e2) { lista = []; } } const nuovaLista = lista.filter(el => el[corpo.matchField] !== corpo.matchValue); const nuovoValore = JSON.stringify(nuovaLista); if (riga === -1) sheet.appendRow([corpo.key, nuovoValore]); else sheet.getRange(riga, 2).setValue(nuovoValore); risultato = { ok: true, value: nuovoValore }; } else { risultato = { error: 'azione POST sconosciuta: ' + corpo.action }; } } catch (err) { risultato = { error: String(err) }; } finally { try { lock.releaseLock(); } catch (e2) {} } return ContentService.createTextOutput(JSON.stringify(risultato)) .setMimeType(ContentService.MimeType.JSON); }
